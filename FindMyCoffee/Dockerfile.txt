FROM mcr.microsoft.com/dotnet/sdk:9.0 AS build
WORKDIR /src

COPY FindMyCoffee.csproj ./
RUN dotnet restore FindMyCoffee.csproj

COPY . ./
RUN dotnet publish FindMyCoffee.csproj -c Release -o /app/publish /p:UseAppHost=false

FROM mcr.microsoft.com/dotnet/aspnet:9.0 AS runtime
WORKDIR /app

COPY --from=build /app/publish ./

ENV ASPNETCORE_ENVIRONMENT=Production

ENTRYPOINT ["dotnet", "FindMyCoffee.dll"]
